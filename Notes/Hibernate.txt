package com.cognizant;

import Student.members;
import org.hibernate.Session;
import org.hibernate.SessionFactory;
import org.hibernate.Transaction;
import org.hibernate.cfg.Configuration;

/**
 * Hello world!
 *
 */
public class
App
{
    public static void main( String[] args )
    {
        members s = new members("192211541","tharun",20); // created a member class and declared fields
        Configuration config = new Configuration(); // setup new configuration
        config.addAnnotatedClass(members.class); // we are telling you handle that class
        config.configure("HibernateConfig.xml"); // specify the hibernate configure class where you configure password, url, username, and driver
        SessionFactory sf = config.buildSessionFactory(); // session is created by session factory 
        Session session = sf.openSession(); // we will open one new session
        Transaction transaction = session.beginTransaction(); // everything is a transaction so all the query should be writern between start and commit
        session.persist(s); // we are pushing the value into table before it used to be save but to maintian the standards hibernated renamed it to persist
        transaction.commit(); // save the changes 
    }
}
